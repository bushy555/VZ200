#include <Bluepad32.h>

const int Z80_D0 = 32;
const int Z80_D1 = 33;
const int Z80_D2 = 25;
const int Z80_D3 = 26;
const int Z80_D4 = 27;
const int Z80_D5 = 14;

const int Z80_A0 = 36;
const int Z80_A1 = 39;
const int Z80_A2 = 34;
const int Z80_A3 = 35;

byte      A00;
byte      A01;
byte      A02;
byte      A03;

byte      VZJoystickByte;


ControllerPtr myControllers[BP32_MAX_GAMEPADS];

// This callback gets called any time a new gamepad is connected.
// Up to 4 gamepads can be connected at the same time.
void onConnectedController(ControllerPtr ctl) {
    bool foundEmptySlot = false;
    for (int i = 0; i < BP32_MAX_GAMEPADS; i++) {
        if (myControllers[i] == nullptr) {
            // Additionally, you can get certain gamepad properties like:
            // Model, VID, PID, BTAddr, flags, etc.
            ControllerProperties properties = ctl->getProperties();
            myControllers[i] = ctl;
            foundEmptySlot = true;
            break;
        }
    }
}

void onDisconnectedController(ControllerPtr ctl) {
    bool foundController = false;

    for (int i = 0; i < BP32_MAX_GAMEPADS; i++) {
        if (myControllers[i] == ctl) {
            myControllers[i] = nullptr;
            foundController = true;
            break;
        }
    }
}

void processGamepad(ControllerPtr ctl) {

    if (ctl->dpad() & 0x01) VZJoystickByte &= 0x3E;   // up 
    if (ctl->dpad() & 0x02) VZJoystickByte &= 0x3D;   // down
    if (ctl->dpad() & 0x04) VZJoystickByte &= 0x37;   // right 
    if (ctl->dpad() & 0x08) VZJoystickByte &= 0x3B;   // left
    
    if ((ctl->buttons() & 0x0001) && (A01 == LOW)) VZJoystickByte  = 0x2F;   // arm     button B     
    if ((ctl->buttons() & 0x0004) && (A00 == LOW)) VZJoystickByte &= 0x2F;   // fire    button Y
}


// Arduino setup funcion. Runs in CPU 1
void setup() {

    pinMode(Z80_D5, OUTPUT);  
    pinMode(Z80_D4, OUTPUT);  
    pinMode(Z80_D3, OUTPUT);  
    pinMode(Z80_D2, OUTPUT);  
    pinMode(Z80_D1, OUTPUT);  
    pinMode(Z80_D0, OUTPUT);  
    
    pinMode(Z80_A0, INPUT);  
    pinMode(Z80_A1, INPUT);  
    pinMode(Z80_A2, INPUT);  
    pinMode(Z80_A3, INPUT);  
  
    const uint8_t* addr = BP32.localBdAddress();

    // Setup the Bluepad32 callbacks
    BP32.setup(&onConnectedController, &onDisconnectedController);

    // "forgetBluetoothKeys()" should be called when the user performs
    // a "device factory reset", or similar.
    // Calling "forgetBluetoothKeys" in setup() just as an example.
    // Forgetting Bluetooth keys prevents "paired" gamepads to reconnect.
    // But might also fix some connection / re-connection issues.
    BP32.forgetBluetoothKeys();

    // Enables mouse / touchpad support for gamepads that support them.
    // When enabled controllers like DualSense and DualShock4 generate two connected devices:
    // - First one: the gamepad
    // - Second one, which is a "vritual device", is a mouse
    // By default it is disabled.
    //BP32.enableVirtualDevice(false);
}

// Arduino loop function. Runs in CPU 1
void loop() {

      VZJoystickByte = 0x3F;    // no buttons pressed, return byte is 63 decimal

      A00 = digitalRead(Z80_A0);
      A01 = digitalRead(Z80_A1); 
      A02 = digitalRead(Z80_A2);
      A03 = digitalRead(Z80_A3); 


    // This call fetches all the gamepad info from the NINA (ESP32) module.
    // Just call this function in your main loop.
    // The gamepads pointer (the ones received in the callbacks) gets updated
    // automatically.
    BP32.update();


    // It is safe to always do this before using the gamepad API.
    // This guarantees that the gamepad is valid and connected.
    for (int i = 0; i < BP32_MAX_GAMEPADS; i++) {
        ControllerPtr myController = myControllers[i];

        if (myController && myController->isConnected()) {
            if (myController->isGamepad()) {
                processGamepad(myController);
            } else {
                continue;
            }
        }
    }

    digitalWrite(Z80_D5, (VZJoystickByte & 0x20));
    digitalWrite(Z80_D4, (VZJoystickByte & 0x10));
    digitalWrite(Z80_D3, (VZJoystickByte & 0x08));
    digitalWrite(Z80_D2, (VZJoystickByte & 0x04));
    digitalWrite(Z80_D1, (VZJoystickByte & 0x02));
    digitalWrite(Z80_D0, (VZJoystickByte & 0x01));

    
    // The main loop must have some kind of "yield to lower priority task" event.
    delay(5);
}
